# 0. Pin every revision (network; writes part211/checkpoints.json). Do not run before the design is registered.

# 1. Draw the two text sets, once per family's tokenizer (network; needs `datasets` and the tokenizers). The row list
#    is written once, so both families draw from the same rows in the same order (the second call reads --rows).
/home/elisepc/neuron-probe/.venv/bin/python -I -B part211/text211.py --tokenizer EleutherAI/pythia-160m \
    --rows-out results/part211/rows.json --out results/part211/texts_pythia.json

    --rows results/part211/rows.json --out results/part211/texts_olmo.json

# 2. The pilot (design: Size): R0 at steps 0 and 143000 and W1 at step 143000, set A; stop rules; a stop is a HOLD.

/home/elisepc/neuron-probe/.venv/bin/python -I -B part211/run211.py --stage pilot \
    --texts results/part211/texts_pythia.json --out results/part211

# 3. The main run, resumable (reference checkpoints first, both sets, then the rest; complete records are skipped).
/home/elisepc/neuron-probe/.venv/bin/python -I -B part211/run211.py --stage main \
    --texts results/part211/texts_pythia.json --out results/part211
/home/elisepc/neuron-probe/.venv/bin/python -I -B part211/run211.py --stage main \
    --texts results/part211/texts_olmo.json --out results/part211

# 4. Score: reads only the distance files.
